#include <Common.Vhlsl.txt>
struct VS_INPUT		{
	float4 Position	: POSITION;
	float2 TexCoord1: TEXCOORD1;
};
struct VS_OUTPUT		{
	float4 Position		: POSITION;
	float2 TcGradientV	: TEXCOORD0;
	float2 TcHorizon	: TEXCOORD1;
};
const float4x4	GbxVisualPrCamera;
const float4x2	GbxVPositionToTexCoord0;
const float4	GbxVisualMadBBox01[2];

vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;
	Output.Position		= mul(v.Position, GbxVisualPrCamera);
	Output.TcGradientV	= mul(v.Position, GbxVPositionToTexCoord0);
	Output.TcHorizon 	= v.TexCoord1;
	return Output;
}
