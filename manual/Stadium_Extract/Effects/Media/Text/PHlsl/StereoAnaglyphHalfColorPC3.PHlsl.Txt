#include <Common.Phlsl.txt>
struct PS_INPUT		{
	float2 TcLeft	: TEXCOORD0;
	float2 TcRight	: TEXCOORD1;
};

struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

sampler2D	MapLeft;
sampler2D	MapRight;

const fpart In01GrayToFullColor = 0.5f;

fpart3 GrayToHalf(fpart3 _Color)
{
#if 0
	fpart  Lumi = dot(_Color, fpart3(1/2.0, 1/4.0, 1/4.0));
#else
	fpart  Lumi = dot(_Color, fpart3(1/3.0, 1/3.0, 1/3.0));
#endif
	fpart3 Half;
	Half.r = Lumi;
	if (_Color.g+_Color.b<0.05f) {
		Half.gb = Lumi;
	} else if (_Color.g>_Color.b) {
		Half.g = 2*Half.r/(1+_Color.b/_Color.g);
		Half.b = Half.g*_Color.b/_Color.g;
	} else {	
		Half.b = 2*Half.r/(1+_Color.g/_Color.b);
		Half.g = Half.b*_Color.g/_Color.b;
	}
	return lerp(Lumi, Half, In01GrayToFullColor);
}

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	Output.Color	= fpart4(1,0,0,1);

	fpart4 Left = tex2D(MapLeft, v.TcLeft);
	fpart4 Right= tex2D(MapRight,v.TcRight);

	fpart3 Red = GrayToHalf(Left);
	fpart3 Cyan = GrayToHalf(Right);

	Output.Color.r	= Red.r;
	Output.Color.gb	= Cyan.gb;
	Output.Color.a  = 0.5*(Left.a+Right.a);
	return Output;
}
