sampler2D	MapColor;
sampler2D	MapDepthBlur;

#include <Common.PHlsl.txt>
struct PS_INPUT		{
	float2 TexCoord	: TEXCOORD0;
};
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

const float		MaxCoC = 4;
const float2 	FilterTaps[DGbxPMrtDepthBlurPC3_TapCount];

static inline fpart4 DoMrtDOF(float2 _TexCoord)
{
	// Get center sample
	fpart4 	ColorSum = tex2D(MapColor, _TexCoord);
	fpart2 	CenterDepthBlur = tex2D(MapDepthBlur, _TexCoord);

	// Compute CoC size based on blurriness
	float 	SizeCoC = CenterDepthBlur.y * MaxCoC;

	float TotalWeight = 1.0f;

	// Run through all taps
	for (int i = 0; i < DGbxPMrtDepthBlurPC3_TapCount; i++)
	{
		// Compute tap coordinates
		float2 TapCoord = _TexCoord + FilterTaps[i] * SizeCoC;

		// Fetch tap sample
		fpart4 TapColor = tex2D(MapColor, TapCoord);
		float2 TapDepthBlur = tex2D(MapDepthBlur, TapCoord);

		// Compute tap contribution
		float TapWeight = (TapDepthBlur.x > CenterDepthBlur.x) ? 1.0f : TapDepthBlur.y;

		// Accumulate color and contribution
		ColorSum 	+= TapColor * TapWeight;
		TotalWeight	+= TapWeight;
	}

	// Normalize to get proper luminance
	return ColorSum / TotalWeight;
}

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;

	Output.Color	= DoMrtDOF(v.TexCoord);
//////////////////////

//	Output.Color	= Output.Color*0.001 + DepthBlur.y;
	return Output;
}
