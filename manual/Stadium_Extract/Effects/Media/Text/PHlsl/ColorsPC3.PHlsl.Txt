#include <Common.Phlsl.txt>
struct PS_INPUT		{
	float2 TexCoord	: TEXCOORD0;
};
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

sampler2D	MapColor;
sampler2D	MapDepthInX;

const fpart4x3	LinearTransfo = fpart4x3(
	1,0,0,
	0,1,0,
	0,0,1,
	0,0,0
);
const fpart4x3	LinearTransfoZFar = fpart4x3(
	1,0,0,
	0,1,0,
	0,0,1,
	0,0,0
);
const fpart3	ContrastPower = fpart3(1,1,1);
const fpart3	ContrastPowerZFar = fpart3(1,1,1);
const fpart2	Depth_Scale_Bias = fpart2(1,0);

#define DLinearTransfo
#define DDepthLinear
#define DContrast
#define DDepthContrast

fpart3	Contrast(fpart3 _Input, fpart3 _Power)
{
	bool3	IsAboveHalf = _Input > 0.5 ;
	fpart3	ToRaise	= saturate(2*(IsAboveHalf ? 1-_Input : _Input));
	fpart3	Output	= 0.5*pow(ToRaise, _Power);
	Output = IsAboveHalf ? 1-Output : Output;
	return Output;
}

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	fpart4		ColorIn = tex2D(MapColor, v.TexCoord);
	fpart4		ColorOut = ColorIn;
	fpart		Depth = tex2D(MapDepthInX, v.TexCoord).r;
	Depth = saturate(Depth*Depth_Scale_Bias.x + Depth_Scale_Bias.y);	

	ColorOut.a	= 1;
#ifdef DContrast
#ifdef DDepthContrast
	fpart3		ContrastAtZ = lerp(ContrastPower, ContrastPowerZFar, Depth);
#else
	fpart3		ContrastAtZ = ContrastPower;
#endif
	ColorOut.rgb	= Contrast(ColorIn.rgb, ContrastAtZ);
#endif
	ColorIn = ColorOut;
#ifdef DLinearTransfo
	ColorOut.rgb	= mul(ColorIn, LinearTransfo);
#endif
#ifdef DDepthLinear
	fpart3		RgbAtZFar = mul(ColorIn, LinearTransfoZFar);
	ColorOut.rgb	= lerp(ColorOut.rgb, RgbAtZFar, Depth);
#endif
	Output.Color.rgb	= ColorOut;
	return Output;
}
