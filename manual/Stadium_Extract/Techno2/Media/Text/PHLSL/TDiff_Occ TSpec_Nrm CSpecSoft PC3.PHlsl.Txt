#include <Common.PHlsl.txt>
struct PS_INPUT		{
	float4 NormalLow_IsLight	: COLOR0;
	float3 PreLight		: COLOR1;
	float4 Tc0_Tc1		: TEXCOORD0;	// .xywz (ps_2_0 optim)
	float3 TgtToWorldX 	: TEXCOORD1;
	float3 TgtToWorldY	: TEXCOORD2;
	float3 TgtToWorldZ	: TEXCOORD3;
	float3 EyeToPos		: TEXCOORD4;
	float2 TcClouds		: TEXCOORD5;
	float2 TcLightGenP	: TEXCOORD6;
	float4 TcShadow		: TEXCOORD7;
};
#define TcShadowBuffer TcShadow

struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

// DTwkPSkipMap_PreLightGen	(not parsed in includes)
// DTwkPSkipMap_Normal
// DTwkPQualityLess_High

#define DIncUniform
#if		defined(DIncUniform) && !defined(DIncCode)

sampler2D	MapDiffuse;
sampler2D	MapSpecular;
sampler2D	MapNormal;
sampler2D	MapOcclusion;
samplerCUBE MapCubeAmbient;
samplerCUBE MapReflectSoft;
sampler2D	MapFresnel;
sampler2D	MapClouds;
sampler2D	MapPreLightGen;

const fpart4	GbxLightDirDirInWorld0;
const fpart4	GbxLightGenP_ScaleAD_TransDA;

const fpart 	SpecularPower = 20;
const fpart4	In01Comp1PreMod_ReflectSoft = fpart4(0.5,0,0,1); 
const fpart4	In01Comp1PreMod_DirSpecular = fpart4(0.3,0,0,1);

DGbxPGenLoadPC3_Shadow

void PreMod(in out fpart3 _Dst, in fpart4 _PreMod, in fpart _Intens)
{
	_Dst *= (_PreMod.w - _Intens*_PreMod.x);
}

#elif	defined(DIncCode)

#ifdef DTwkPQualityLess_High
#define DTwkPSkipMap_Normal 1
#endif

#ifdef DTcDiffuse	// Mapper tweak
	fpart4	Diffuse 		= tex2D(MapDiffuse, 	DTcDiffuse);
	fpart4	Specular 		= tex2D(MapSpecular, 	DTcSpecular);
	fpart3	NormalInTgt01 	= tex2D(MapNormal, 		DTcNormal).agb;
	fpart3	Clouds 			= tex2D(MapClouds, 		v.TcClouds).rgb;
	fpart	Occlusion 		= tex2D(MapOcclusion, 	DTcOcclusion).x;
#else				// Sampling tweak (Transition support)
	fpart3	Clouds 			= tex2D(MapClouds, 		DTcClouds);
	fpart4	Diffuse, Specular;
	fpart3	NormalInTgt01;
	fpart	Occlusion;
	MSample_Diff_Spec_NrmInTgt01_Occ(Diffuse, Specular, NormalInTgt01, Occlusion);
#endif
	fpart3	NormalInTgt 	= NormalInTgt01*2 - (254.0/255);
	fpart3	NormalLow = v.NormalLow_IsLight.xyz*2-1;
	fpart	IsLighted = v.NormalLow_IsLight.a;

#ifdef DTwkPSkipMap_Normal
	fpart3	nNormal = normalize(NormalLow);
#else
	fpart3	TgtToWorldX = v.TgtToWorldX;
	fpart3	TgtToWorldY = v.TgtToWorldY;
	fpart3	TgtToWorldZ = v.TgtToWorldZ;

//NormalInTgt = NormalInTgt*0.01 + float3(0,0,1);

	fpart3	nNormal = float3(
		dot(NormalInTgt, TgtToWorldX),
		dot(NormalInTgt, TgtToWorldY),
		dot(NormalInTgt, TgtToWorldZ)
	);
	nNormal = normalize(nNormal);
//nNormal = nNormal*0.01 + float3(0,0,1);

#endif

	fpart4	Ambient = texCUBE(MapCubeAmbient, nNormal);

	// AmbientCube stores directionnal light intens in alpha:
	Ambient.a = max(0,dot(nNormal, -GbxLightDirDirInWorld0.xyz));

	fpart3	LDiffusDir = (Ambient.a*ShadowDiffus*GbxLightDirRgb0*IsLighted).rgb;

//Occlusion = Occlusion*0.001 + 1;

#ifndef DTwkPSkipMap_PreLightGen
	fpart3	LightGenP		= tex2D(MapPreLightGen, v.TcLightGenP).rgb;
/* does not work : ça coupe le dégradé d'ambient occ.
#ifdef DGbxPDefinedPC3_Shadow	
//	fpart	DirSoftInside = saturate(2*LightGenP.g);
	fpart	DirSoftInside = saturate(2*LightGenP.g + GbxDbgTest0.x);
	LightGenP.r *= DirSoftInside;
#endif
*/
	LightGenP.rg = LightGenP.rg*GbxLightGenP_ScaleAD_TransDA.xy+GbxLightGenP_ScaleAD_TransDA.wz;
#ifndef DGbxPDefinedPC3_Shadow
	LDiffusDir	*= LightGenP.g;
#endif	// combination don't work well, especially because of cars
	Ambient 	*= LightGenP.r;
#endif

	fpart3	LDiffus = Ambient.rgb;
#ifndef DTwkPSkipMap_Occlusion
	LDiffus *= Occlusion;
#endif
	LDiffus += LDiffusDir;
//	LDiffus = saturate(LDiffus);
	LDiffus = saturate(LDiffus);

	fpart3	EyeToPos = normalize(v.EyeToPos);
	fpart3	Reflect = reflect(EyeToPos, nNormal);

	fpart4	ReflectSoft_Spec = texCUBE(MapReflectSoft, Reflect);
	fpart3	ReflectSoft = ReflectSoft_Spec.rgb;

#ifndef DTwkPSkipMap_PreLightGen
	ReflectSoft	*= LightGenP.r;
#endif

	fpart2	TcFresnel;
	TcFresnel.x = dot(-EyeToPos, nNormal);
	TcFresnel.y = Specular.a;
	fpart3	Fresnel = tex2D(MapFresnel, TcFresnel).rgb;

// Remove fresnel where normal map is back reflecting (compared to low normal)
//	fpart	FresnelOnlyFrontReflect = dot(Reflect, NormalLow)>0;
	fpart	FresnelOnlyFrontReflect = saturate((dot(Reflect, NormalLow)+0.3)*4);

	Fresnel *= FresnelOnlyFrontReflect;

	fpart3	LSpecularSoft = Fresnel*ReflectSoft*Specular.rgb;

//	fpart	SpecIntens = pow(max(0,dot(Reflect, -GbxLightDirDirInWorld0)), SpecularPower);
	fpart	SpecIntens = ReflectSoft_Spec.a;
	SpecIntens *= IsLighted;
	SpecIntens *= ShadowSpecular;
	SpecIntens *= Specular.a;
#ifndef DTwkPSkipMap_PreLightGen
	SpecIntens *= LightGenP.g;
#endif

	fpart3	LSpecular = (SpecIntens*GbxLightDirRgb0*Specular).rgb;

	Output.Color.rgb = LDiffus*Diffuse.rgb;
#ifndef DTwkPSkipMap_Clouds
	Output.Color.rgb*= Clouds*2;
#endif

	// Add ReflectSoft after pre-modulating with reverse intens
	fpart	RelfectSoftI = dot(LSpecularSoft, fpart3(1,1,1)*rsqrt(3.0h));
	PreMod(Output.Color.rgb, In01Comp1PreMod_ReflectSoft, RelfectSoftI);
	Output.Color.rgb += LSpecularSoft;

	// Add Specular after pre-modulating with reverse intens
	PreMod(Output.Color.rgb, In01Comp1PreMod_DirSpecular, SpecIntens);
	Output.Color.rgb += LSpecular;
//	Output.Color.a		= 1;
	Output.Color.a = Diffuse.a;

#ifdef DGbxEditor
#define MDebug(__Value) Output.Color.rgb = Output.Color.rgb*0.001 + (__Value);
#ifndef DTwkPSkipMap_Clouds
	MDbg_Clouds(			Clouds);
#endif
	MDbg_Diffuse(			LDiffus*Diffuse);
	MDbg_Specular(			LSpecular);
	MDbg_Fresnel(			LSpecularSoft);
	MDbg_NormalInW(			nNormal*0.5+0.5);
#ifndef DTwkPSkipMap_Normal
	MDbg_TangentUW(			0.5+0.5*float3(v.TgtToWorldX.x, v.TgtToWorldY.x, v.TgtToWorldZ.x))
#endif
	MDbg_MapDiffuse(		Diffuse);
	MDbg_MapSpecular(		Specular);
	MDbg_MapOcclusion(		Occlusion);
	MDbg_LightDiffuse(		LDiffus);
	MDbg_LightDiffuseAmb(	Occlusion*Ambient);
	MDbg_LightDiffuseDir0(	Occlusion*LDiffusDir);
#ifdef DTwkPSkipMap_PreLightGen
	MDbg_Occlusion(			Occlusion);
#else
	MDbg_Occlusion(			Occlusion*LightGenP.r);
#endif
#if DGbxPDefinedPC3_Shadow
	MDbg_MapperShadow0(		DGbxPGenCodePC3_ShadowDbgMapper0);
#endif
#endif

#else
float2 f = ERROR_MISSING_DEFINE;
#endif
