#include <Common.Vhlsl.txt>
struct VS_INPUT		{
	float4 Position		: POSITION;
	float3 Normal		: NORMAL;
	float4 PreLight		: COLOR0;
#ifdef DTwkVSkipMap_Occlusion
	float4 TcPreLightGen	: TEXCOORD0;// Expand to (u,v,0,1) to transform with float4x2...
#else
	float2 TcOcclusion	: TEXCOORD0;
	float4 TcPreLightGen	: TEXCOORD1;// Expand to (u,v,0,1) to transform with float4x2...
#endif
};
struct VS_OUTPUT		{
	float4 Position		: POSITION;
	float  Fog			: FOG;
	float4 Color0		: COLOR0;
	float4 Color1		: COLOR1;
	float2 TcBlend1		: TEXCOORD0;
	float2 TcBlend2		: TEXCOORD1;
	float2 TcBlendI		: TEXCOORD2;
#ifndef DTwkVSkipMap_Clouds
	float2 TcClouds		: TEXCOORD3;
#endif
#ifndef DTwkVSkipMap_Occlusion
	float2 TcOcclusion	: TEXCOORD4;
#endif
#ifndef DTwkVSkipMap_PreLightGen
	float2 TcPreLightGen: TEXCOORD5;
#endif
#ifdef DGbxVDefinedPC3_Shadow
	float4 TcShadowBuffer: TEXCOORD6;
#endif
};

const float4x4	GbxVisualPrCamera;
const float4x3	GbxVisualToWorld;
const float4	GbxEyeInVisual;
const float4x2	GbxVPositionToTexCoord_MapBlend1;
const float4x2	GbxVPositionToTexCoord_MapBlend2;
const float4x2	GbxVPositionToTexCoord_MapBlendI;
const float4x2	GbxVPositionToTexCoord_MapClouds;
const float4x2	GbxVTexCoordToLightGenP;

DGbxVGenLoadPC3_Shadow

vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;
	Output.Position		= mul(v.Position, GbxVisualPrCamera);
	Output.Fog 			= mul(v.Position, GbxVisualToFogTnL);
	Output.TcBlend1.xy	= mul(v.Position, GbxVPositionToTexCoord_MapBlend1);
	Output.TcBlend2		= mul(v.Position, GbxVPositionToTexCoord_MapBlend2);
	Output.TcBlendI		= mul(v.Position, GbxVPositionToTexCoord_MapBlendI);
#ifndef DTwkVSkipMap_Clouds
	Output.TcClouds		= mul(v.Position, GbxVPositionToTexCoord_MapClouds);
#endif
#ifndef DTwkVSkipMap_Occlusion
	Output.TcOcclusion	= v.TcOcclusion;
#endif
	DGbxVGenCodePC3_Shadow
#ifdef DTwkVSkipMap_PreLightGen
	DGbxVGenCodePC3_Prelight_NoVertexColor
	Output.Color0.a		= dot(v.Normal,1); // force normal usage
	Output.Color1.a		= dot(v.Normal,1); // force normal usage
#else
	Output.TcPreLightGen= mul(v.TcPreLightGen, GbxVTexCoordToLightGenP);
	Output.Color0.rgb	= GbxLightDirRgb0*max(0, dot(v.Normal, -GbxLightDirDir0));
	// pass Normal to pixel, else it may be skipped in exported geometry
	// and PreLightGenerator 'll bug !!!
#endif
	return Output;
}
