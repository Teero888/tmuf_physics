#include <Common.VHlsl.txt>
struct VS_INPUT {
	float4 Position 	: POSITION0;
	float3 Normal 		: NORMAL0;
	float4 Color0		: COLOR0;
	float2 TcDiffuse 	: TEXCOORD0;
	float2 TcOcclusion	: TEXCOORD1;
};
struct VS_OUTPUT {
	float4 Position		: POSITION0;
	float1 Fog			: FOG;
	float4 Color0		: COLOR0;
	float4 Fresnel		: COLOR1;
	float2 TcDiffuse	: TEXCOORD0;
	float2 TcOcclusion	: TEXCOORD1;
	float2 TcClouds		: TEXCOORD2;
	float3 EyeToPosInW	: TEXCOORD3;
};
const float4x4 	GbxVisualPrCamera;
const float4x2 	GbxVPositionToTexCoord2;
const float4	GbxEyeInVisual;
const float4x3	GbxVisualToWorld;

vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT Output = (VS_OUTPUT)0;
#ifdef DGbxVDefinedPC3_Prelight
	DGbxVGenCodePC2_Prelight
#else
	Output.Color0.rgb 	= LightAD_Dir1(v.Normal);
#endif
	Output.Color0.a  	 = 1;
	Output.Position = mul(v.Position, GbxVisualPrCamera);
	Output.Fog			= dot(v.Position, GbxVisualToFogTnL);
	Output.TcDiffuse 	= v.TcDiffuse;
	Output.TcOcclusion	= v.TcOcclusion;
	Output.TcClouds		= mul(v.Position, GbxVPositionToTexCoord2);
	float3	EyeToPosInV = v.Position-GbxEyeInVisual;
	Output.EyeToPosInW	= mul(EyeToPosInV, GbxVisualToWorld);
	float3	nEyeToPosInV = normalize(EyeToPosInV);
	float3	FresnelRgb = float3(0.20, 0.20, 0.30);
	Output.Fresnel.a	= pow(max(0,1-dot(-nEyeToPosInV, v.Normal)),5);
	Output.Fresnel.rgb	= Output.Fresnel.a*FresnelRgb;
	Output.Fresnel.a	= lerp(0.05, 0.5, Output.Fresnel.a);
	return Output;
};
