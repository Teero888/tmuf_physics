
#include <Common.PHlsl.txt>

struct PS_INPUT		{
	float2 TcDiffuse	: TEXCOORD0;
	float2 TcClouds		: TEXCOORD1;
	fpart3 NormalInW	: TEXCOORD2;
	float3 PosInW		: TEXCOORD3;
	float4 TcShadow		: TEXCOORD4;
	float2 TcLightFromMap: TEXCOORD5;
	fpart4 Color0		: COLOR0;
	fpart4 Color1		: COLOR1;
};
#define TcShadowBuffer TcShadow

struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

sampler2D   MapClouds;
sampler2D   MapDiffuse_Gloss;
sampler2D	MapHemiSpec;
samplerCUBE	MapCubeInvHemi;
sampler2D    MapSelfIllum;
sampler2D	MapLightFromMap;
sampler2D	MapDirtMarks;

DGbxPGenLoadPC3_Shadow

const fpart4	GbxHemiTexToSub_Sxy_Tyx_MapHemiSpec;
const fpart4x3	GbxHemiWorldToEye_MapHemiSpec;

const fpart2	GbxLightFromMap_ScaleRGB_TransRGB;

// ascending comp: to remove
#define GbxVehicle_DirtIntens_Gloss GbxVehicle_DirtIntens
const fpart2	GbxVehicle_DirtIntens_Gloss;

//const float3		GbxLightAmbient;
//const float3		GbxLightDirDirInWorld0;
//const float3		GbxLightDirRgb0;

// Renaming
#define	DHemiTexToSub_Sxy_Tyx GbxHemiTexToSub_Sxy_Tyx_MapHemiSpec
#define DHemiWorldToEye GbxHemiWorldToEye_MapHemiSpec

PS_OUTPUT Main(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	DGbxPGenCodePC3_Shadow

	fpart4		DiffuseGloss = tex2D(MapDiffuse_Gloss, v.TcDiffuse);
	fpart4		SelfIllum = tex2D(MapSelfIllum, v.TcDiffuse);
	fpart3		NormalInW 	= v.NormalInW;//normalize(v.NormalInW);
	float3		PosInW	= v.PosInW;//normalize(v.PosInW);
	fpart3		LDiffus = lerp(v.Color0,v.Color1,ShadowDiffus);
	fpart3		ReflecInW = normalize(PosInW - 2*dot(NormalInW,PosInW)*NormalInW);
	fpart4		LightFromMap = tex2D(MapLightFromMap, v.TcLightFromMap);

#ifndef DTwkPSkipMap_DirtMarks
	fpart4		DirtMarks = tex2D(MapDirtMarks, v.TcDiffuse);
	fpart		DirtIntens = DirtMarks.a*GbxVehicle_DirtIntens_Gloss.x;
	DiffuseGloss.rgb = lerp(DiffuseGloss, DirtMarks, DirtIntens);
	DiffuseGloss.a	 = lerp(DiffuseGloss, GbxVehicle_DirtIntens_Gloss.y, DirtIntens).a;
#endif

	///////////////////////
	// Complex mapping : compute pixel's reflection vector
	fpart3		ReflecInE = mul(ReflecInW, DHemiWorldToEye);
	fpart2		TcHemiSpec = texCUBE(MapCubeInvHemi, ReflecInE);

	TcHemiSpec = TcHemiSpec*DHemiTexToSub_Sxy_Tyx.xy + DHemiTexToSub_Sxy_Tyx.wz;

	fpart4		HemiSpec = tex2D(MapHemiSpec, TcHemiSpec);
	fpart3		SpecSparkles = HemiSpec.rgb;

#ifndef DTwkPSkipMap_LightFromMap
	LightFromMap*= GbxLightFromMap_ScaleRGB_TransRGB.x;
	LightFromMap+= GbxLightFromMap_ScaleRGB_TransRGB.y;
	LDiffus *= LightFromMap;
#endif

//	DiffuseGloss.a 		= 1; 
//	DiffuseGloss.rgb 	= DiffuseGloss*0.0001 +1 ;

	float3	LHdr = (fpart3)GbxLightAmbient+LDiffus;

	Output.Color.a		= 0;
	Output.Color.rgb	= DiffuseGloss*LHdr;
	Output.Color.rgb   += DiffuseGloss.a*SpecSparkles;
	Output.Color.rgb   *= 2*(fpart3)tex2D(MapClouds, v.TcClouds);
	Output.Color.rgb   += SelfIllum;

///////// DEBUG //////////////////
//	Output.Color.rgb	= Output.Color.rgb*0.001 + IHdr ;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LHdr ;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LDiffus;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LDiffus*(1-Metal);
//	Output.Color.rgb	= Output.Color.rgb*0.001 + (fpart3)GbxLightAmbient;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + (fpart3)GbxLightAmbient+LDiffus;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + (fpart3)GbxLightAmbient+LDiffus*(1-Metal);
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LDiffus*DiffuseGloss;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + DiffuseGloss;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + DiffuseGloss.a;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + SpecSparkles;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + SpecVarnish;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + DiffuseGloss*Metal*SpecSparkles;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + Gloss;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + Metal;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LightFromMap;
	return Output;
}

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);}
ps_2_a PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);}
