	float4 Position		: POSITION;
	float3 Normal		: NORMAL;
	float2 TcDiffuse	: TEXCOORD0;
};

	float4 Position		: POSITION;
	float1 Fog			: FOG;
	float2 TcDiffuse	: TEXCOORD0;
	float2 TcClouds		: TEXCOORD1;
	float3 NormalInW	: TEXCOORD2;
	float3 EyeToPosInW	: TEXCOORD3;
#ifdef DGbxVDefinedPC3_Shadow
	float4 TcShadow		: TEXCOORD4;
#endif
#ifndef DTwkVSkipMap_LightFromMap
	float2 TcLightFromMap	: TEXCOORD5;
#endif
/*
#ifdef DUseHdr
	float4 RgbDiffuse	: TEXCOORD5;	// TODO : 16b...
#else
	float4 RgbDiffuse	: COLOR0;
#endif
	// Debug purpose
	float4 RgbDiffuseHdr: TEXCOORD5;	// TODO : 16b...
*/
	float4 Color0	: COLOR0;
	float4 Color1	: COLOR1;
};
#define TcShadowBuffer TcShadow

const float4x4	GbxVisualPrCamera;
const float4x3	GbxVisualToWorld;
const float4x3	GbxVisualToEye;
const float4	GbxEyeInVisual;
const float4x2	GbxVPositionToTexCoord0;
const float4	GbxGameUnder_BlendIn_Fade_FOut_FIn;
const float4x2	GbxVPositionTo01LightFromMap;

float4	RgbLightUnderground = float4(240.f/255.f, 211.f/255.f, 172.f/255.f, 1);

DGbxVGenLoadPC3_Shadow

vs_2_0 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;
	
	DGbxVGenCodePC3_Shadow

	float4	LDiffus = 0;

//	LDiffus += GbxLightAmbient;

//LDiffus *= 0.001;

	// Big issue : activating both loops made a vertex shader run error (in soft, corrupt in hard)
	// Direct3D9: (ERROR) :ERROR: Vertex shader relative address out-of-bounds!
	// But disappeared when I split the light arrays into light.register arrays...

	for (int iBall=0; iBall<GbxLight8BallCount; iBall++) {
		LDiffus.rgb += DoLightBall8(iBall, v.Position, v.Normal);
	}
	for (int iSpot=0; iSpot<GbxLight8SpotCount; iSpot++) {
		LDiffus.rgb += DoLightSpot8(iSpot, v.Position, v.Normal);
	}

//LDiffus *= 0.5;

	float4	LDiffusDir0 = GbxLightDirRgb0*saturate(dot(v.Normal,-GbxLightDirDir0));
	LDiffusDir0 *= GbxGameUnder_BlendIn_Fade_FOut_FIn.z;	// fade instead of blend, else scene shadow 'll pop

	float4	LDiffusUnder = RgbLightUnderground*saturate(v.Normal.y);
	LDiffusUnder *= GbxGameUnder_BlendIn_Fade_FOut_FIn.x;

	float3	EyeToPosInV = v.Position-GbxEyeInVisual;

	Output.Position		= mul(v.Position, GbxVisualPrCamera);
	Output.Fog			= dot(v.Position, GbxVisualToFogTnL);
	Output.Color0		= LDiffus+LDiffusUnder;
	Output.Color1		= LDiffus+LDiffusUnder+LDiffusDir0;
	Output.TcDiffuse	= v.TcDiffuse;
	Output.TcClouds		= mul(v.Position, GbxVPositionToTexCoord0);
	Output.NormalInW	= mul(v.Normal, GbxVisualToWorld);
	Output.EyeToPosInW	= mul(EyeToPosInV, GbxVisualToWorld);
#ifndef DTwkVSkipMap_LightFromMap
	Output.TcLightFromMap = mul(v.Position, GbxVPositionTo01LightFromMap);
#endif
	return Output;
}
