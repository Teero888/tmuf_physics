#include <Common.Vhlsl.txt>
struct VS_INPUT		{
	float4 Position	: POSITION;
	float4 Color0	: COLOR0;
};
struct VS_OUTPUT		{
	float4 HPosition: POSITION;
	float4 Color0	: COLOR0;
	float2 TcBump	: TEXCOORD0;
	float4 TcRefrac	: TEXCOORD1;
	float4 TcReflec	: TEXCOORD2;
	float2 TcFresnel: TEXCOORD3;
};
const float4x4	GbxVisualPrCamera;
const float4x2	GbxVPositionToTexCoord0;
const float4x3	GbxVPositionToTexCrd3d1;
const float4x3	GbxVPositionToTexCrd3d2;
const float4x2	GbxVPositionToTexCoord3;
const float4x3	GbxVisualToWorld;
const float4	GbxTime;
const float		BumpScaleUV = 0.05f;
vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;
	Output.HPosition	= mul(v.Position, GbxVisualPrCamera);
	Output.Color0.rgb	= v.Color0;
	Output.Color0.a		= 0;
//	Output.TcBump		= mul(v.Position, GbxVPositionToTexCoord0);
	Output.TcBump		= mul(v.Position, GbxVisualToWorld).xz*BumpScaleUV;
	Output.TcBump	   += GbxTime.x*0.01;
	Output.TcRefrac		= mul(v.Position, GbxVPositionToTexCrd3d1).xyzz;
	Output.TcReflec		= mul(v.Position, GbxVPositionToTexCrd3d2).xyzz;
	Output.TcFresnel	= mul(v.Position, GbxVPositionToTexCoord3);
	return Output;
}
