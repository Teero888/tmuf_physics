#include <Common.Vhlsl.txt>
struct VS_INPUT		{
	float4 Position	: POSITION;
	float4 Color0	: COLOR0;
};
struct VS_OUTPUT		{
	float4 HPosition: POSITION;
	float4 Color0	: COLOR0;
	float4 TcRefrac	: TEXCOORD0;
	float4 TcReflec	: TEXCOORD1;
	float2 TcFresnel: TEXCOORD2;
};
const float4x4	GbxVisualPrCamera;
const float4x3	GbxVPositionToTexCrd3d0;
const float4x3	GbxVPositionToTexCrd3d1;
const float4x2	GbxVPositionToTexCoord2;
vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;
	Output.HPosition	= mul(v.Position, GbxVisualPrCamera);
	Output.Color0.rgb	= v.Color0;
	Output.Color0.a		= 0;
	Output.TcRefrac		= mul(v.Position, GbxVPositionToTexCrd3d0).xyzz;
	Output.TcReflec		= mul(v.Position, GbxVPositionToTexCrd3d1).xyzz;
	Output.TcFresnel	= mul(v.Position, GbxVPositionToTexCoord2);
	return Output;
}
