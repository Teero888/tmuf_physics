#include <Common.Vhlsl.txt>
#ifndef DTwkVInputUsageTc_PreLightGen
#define DTwkVInputUsageTc_PreLightGen TEXCOORD0
#endif
struct VS_INPUT		{
	float4 Position		: POSITION;
	float3 Normal		: NORMAL;
	float4 TcPreLightGen: DTwkVInputUsageTc_PreLightGen; // expanded to (u,v,0,1) for float4x2 mul
};
struct VS_OUTPUT		{
	float4 Position		: POSITION;
	float3 PosInWorld	: TEXCOORD0;
	float3 NormalInW	: TEXCOORD1;
	float2 TcAccum		: TEXCOORD2;
};

const float4x3	GbxVTexCoordToRaster;
const float4x3	GbxVisualToWorld;

const float4	RasterMadTcAccum = float4(1,1,0,0);

vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;

	Output.Position.xyz= mul(v.TcPreLightGen, GbxVTexCoordToRaster);
	Output.Position.w = 1;
	Output.PosInWorld = mul(v.Position, GbxVisualToWorld);
	Output.NormalInW = mul(v.Normal, GbxVisualToWorld);
	Output.TcAccum = Output.Position.xy*RasterMadTcAccum.xy + RasterMadTcAccum.zw;
	return Output;
}
