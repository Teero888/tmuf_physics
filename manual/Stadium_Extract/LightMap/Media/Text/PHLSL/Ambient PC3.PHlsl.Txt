#include <Common.Phlsl.txt>
#include <Effects\\Media\\Text\\PHlsl\\Lights.Phlsl.Inc.txt>
struct PS_INPUT		{
	float3 PosInWorld	: TEXCOORD0;
	float3 NormalInW	: TEXCOORD1;
};
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

samplerCUBE		MapCubeAmbient;

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	Output.Color.rgb	= texCUBE(MapCubeAmbient, v.NormalInW);
// Output.Color0.rgb	= Output.Color0.rgb*0.001 + v.NormalInW*0.5+0.5;
	Output.Color.a		= 1;
	return Output;
}
