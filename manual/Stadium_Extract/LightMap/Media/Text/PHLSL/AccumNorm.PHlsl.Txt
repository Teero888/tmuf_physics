#include <Common.Phlsl.txt>
struct PS_INPUT		{
	float2 TcLight	: TEXCOORD0;
};
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};
sampler2D		MapLightSSAccum;
const fpart4	GlobalRGB_Power = fpart4(1,1,1,1);
const fpart2	GlobalScaleTrans = fpart2(1,0);
ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	fpart4		SSAccum = tex2D(MapLightSSAccum, v.TcLight);
	if (SSAccum.y>0.01f) {
// Normalization was needed to SSAA
//		Output.Color.rgb	= GlobalColor*SSAccum.x/SSAccum.y;
// MultiLight require only RGB mul, and A=1
		fpart	Value = pow(SSAccum.x, GlobalRGB_Power.a);
		Value = saturate(Value*GlobalScaleTrans.x + GlobalScaleTrans.y);
		Output.Color.rgb	= GlobalRGB_Power.rgb*Value;
		Output.Color.a		= 1;
	}
	else {
		Output.Color		= fpart4(0,0,0,0);
	}
	return Output;
}
