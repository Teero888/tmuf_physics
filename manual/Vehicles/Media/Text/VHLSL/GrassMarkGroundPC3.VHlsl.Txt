#include <Common.Vhlsl.txt>
struct VS_INPUT		{
	float4 Position	: POSITION;
	float4 Color0	: COLOR0;
	float3 Normal	: NORMAL;
	float2 TexCoord0: TEXCOORD0;
};
struct VS_OUTPUT		{
	float4 Position	: POSITION;
	float4 Color0	: COLOR0;
	float2 TcIntens	: TEXCOORD0;
	float2 TcBlendXZ: TEXCOORD1;
	float2 TcClouds	: TEXCOORD2;
#ifdef DGbxVDefinedPC3_Shadow
	float4 TcShadow	: TEXCOORD3;
#endif
};
#define TcShadowBuffer TcShadow
const float4x4	GbxVisualPrCamera;
const float4x2	GbxVPositionToTexCoord1;
const float4x2	GbxVPositionToTexCoord2;

DGbxVGenLoadPC3_Shadow

vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;
	DGbxVGenCodePC3_Shadow
	Output.Position	= mul(v.Position, GbxVisualPrCamera);
//	Output.Color0	= GbxLightDirRgb0*max(0, dot(v.Normal, -GbxLightDirDir0));
	Output.Color0	= v.Color0;
	Output.TcIntens = v.TexCoord0;
	Output.TcBlendXZ= mul(v.Position, GbxVPositionToTexCoord1);
	Output.TcClouds	= mul(v.Position, GbxVPositionToTexCoord2);
	return Output;
}
