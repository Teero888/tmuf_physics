struct VS_INPUT		{
	float4 Position		: POSITION;
	float3 Normal		: NORMAL;
	float2 TcTranslucent: TEXCOORD0;
};
struct VS_OUTPUT		{
	float4 Position		: POSITION;
	float1 Fog			: FOG;
	float4 RgbDiffuse	: COLOR0;
	float2 TcTranslucent: TEXCOORD0;
	float2 TcClouds		: TEXCOORD1;
	float3 NormalInV	: TEXCOORD2;
	float3 EyeVInV		: TEXCOORD3;
};

#include <Common.VHlsl.txt>

const float4x4	GbxVisualPrCamera;
const float4x3	GbxVisualToWorld;
const float4	GbxEyeInVisual;
const float4x2	GbxVPositionToTexCoord0;

vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;
	
	float4	LDiffus = 0;

	LDiffus += GbxLightDirRgb0*saturate(dot(v.Normal,-GbxLightDirDir0));

	Output.Position		= mul(v.Position, GbxVisualPrCamera);
	Output.Fog			= dot(v.Position, GbxVisualToFogTnL);
	Output.RgbDiffuse	= LDiffus;
	Output.TcTranslucent= v.TcTranslucent;
	Output.TcClouds		= mul(v.Position, GbxVPositionToTexCoord0);
	Output.NormalInV	= v.Normal;
	Output.EyeVInV		= v.Position-GbxEyeInVisual;
	return Output;
}
