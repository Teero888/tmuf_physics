#include <Common.Phlsl.txt>
struct PS_INPUT		{
	float4 Color0	: COLOR0;
	float2 TcIntens	: TEXCOORD0;
	float2 TcBlendXZ: TEXCOORD1;
	float2 TcClouds	: TEXCOORD2;
	float4 TcShadow	: TEXCOORD3;
};
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};
#define TcShadowBuffer TcShadow
sampler2D	MapIntens;
sampler2D	MapBlendXZ;
sampler2D	MapClouds;

DGbxPGenLoadPC3_Shadow
ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	DGbxPGenCodePC3_Shadow
	fpart4		Intens = tex2D(MapIntens, v.TcIntens);
	fpart4		BlendXZ= tex2D(MapBlendXZ, v.TcBlendXZ);
	fpart3		Clouds = tex2D(MapClouds, v.TcClouds);
	fpart3		Light = saturate(GbxLightAmbient+ShadowDiffus*v.Color0);
	Output.Color	 = Intens*BlendXZ;
	Output.Color.rgb*= Light;
	Output.Color.rgb*= Clouds*2;
////////////////////////////
//  Debug
//	Output.Color.rgb = Output.Color.rgb*0.001 + Light;
//	Output.Color.rgb = Output.Color.rgb*0.001 + v.Color0;
//	Output.Color.rgb = Output.Color.rgb*0.001 + ShadowDiffus;
	return Output;
}
