
#include <Common.PHlsl.txt>

//#define DUseHdr

struct PS_INPUT		{
	float2 TcDiffuse	: TEXCOORD0;
	float2 TcClouds		: TEXCOORD1;
	fpart3 NormalInW	: TEXCOORD2;
	float3 EyeToPosInW	: TEXCOORD3;
	float4 TcShadow		: TEXCOORD4;
	float2 TcLightFromMap: TEXCOORD5;
#ifdef DUseHdr
	float4 RgbDiffuse	: TEXCOORD5;	// TODO : 16b...
#else
	fpart4 Color0		: COLOR0;
	fpart4 Color1		: COLOR1;
#endif
};
#define TcShadowBuffer TcShadow

struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

sampler2D   MapClouds;
sampler2D   MapDiffuse_Gloss;
sampler2D	MapHemiSpec;
samplerCUBE	MapCubeInvHemi;
samplerCUBE MapEnvCubic;
sampler2D	MapFresnel;
sampler2D	MapLightFromMap;
sampler2D	MapDirtMarks;

DGbxPGenLoadPC3_Shadow

const fpart4	GbxHemiTexToSub_Sxy_Tyx_MapHemiSpec;
const fpart4x3	GbxHemiWorldToEye_MapHemiSpec;

const fpart4	GbxGameUnder_BlendIn_Fade_FOut_FIn;

const fpart2	GbxLightFromMap_ScaleRGB_TransRGB;
const fpart4	GbxLightFromMap_ScaleAD_TransDA;

// ascending comp: to remove
#define GbxVehicle_DirtIntens_Gloss GbxVehicle_DirtIntens
const fpart4	GbxVehicle_DirtIntens_Gloss;


//const float3		GbxLightAmbient;
//const float3		GbxLightDirDirInWorld0;
//const float3		GbxLightDirRgb0;

// Renaming
#define	DHemiTexToSub_Sxy_Tyx GbxHemiTexToSub_Sxy_Tyx_MapHemiSpec
#define DHemiWorldToEye GbxHemiWorldToEye_MapHemiSpec

PS_OUTPUT Main(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	DGbxPGenCodePC3_Shadow

	fpart4		DiffuseGloss = tex2D(MapDiffuse_Gloss, v.TcDiffuse);
	fpart3		NormalInW 	= v.NormalInW;//normalize(v.NormalInW);
	float3		EyeToPosInW	= v.EyeToPosInW;//normalize(v.EyeToPosInW);
	fpart3		LDiffus = lerp(v.Color0,v.Color1,ShadowDiffus);
	fpart3		ReflecInW = normalize(reflect(EyeToPosInW, NormalInW));
	fpart4		EnvCubic = texCUBE(MapEnvCubic, ReflecInW);
	fpart4		LightFromMap = tex2D(MapLightFromMap, v.TcLightFromMap);
	fpart3		Clouds   = tex2D(MapClouds, v.TcClouds);

	EnvCubic *= GbxGameUnder_BlendIn_Fade_FOut_FIn.y;

#ifndef DTwkPSkipMap_DirtMarks
	fpart4		DirtMarks = tex2D(MapDirtMarks, v.TcDiffuse);
	fpart		DirtIntens = DirtMarks.a*GbxVehicle_DirtIntens_Gloss.x;
	DiffuseGloss.rgb = lerp(DiffuseGloss, DirtMarks, DirtIntens);
	DiffuseGloss.a	 = lerp(DiffuseGloss, GbxVehicle_DirtIntens_Gloss.y, DirtIntens).a;
#endif

	///////////////////////
	// Complex mapping : compute pixel's reflection vector
	fpart3		ReflecInE = mul(ReflecInW, DHemiWorldToEye);
/*	fpart4		VecHemiSpec = texCUBE(MapCubeInvHemi, ReflecInE);
	fpart2		TcHemiSpec = mul(VecHemiSpec, WholeToSubTex);*/

	fpart2		TcHemiSpec = texCUBE(MapCubeInvHemi, ReflecInE);

	// Warning 4d->2d split imply source register swizzling, and ps_2_0 supports:
	// .yzwx .zxyw .wzyx only the last use z & w for the first 2 component (but in reverse order...)
	// !! TcHemiSpec*TexToSubScale_Trans.xy + TexToSubScale_Trans.zw; leads to 1 instruction loss !!

	TcHemiSpec = TcHemiSpec*DHemiTexToSub_Sxy_Tyx.xy + DHemiTexToSub_Sxy_Tyx.wz;

	fpart2		TcFresnel = ReflecInE.z;
	fpart		Fresnel = tex2D(MapFresnel, TcFresnel).r;
	fpart4		HemiSpec = tex2D(MapHemiSpec, TcHemiSpec);
	fpart3		SpecSparkles = HemiSpec.rgb;
	fpart3		SpecVarnish = 2*HemiSpec.rgb*HemiSpec.a;
	fpart3		LAmbient = GbxLightAmbient.rgb*VDepAmbOcc;

#ifndef DTwkPSkipMap_LightFromMap
#ifndef DGbxPDefinedPC3_IsNight
	LightFromMap.rg*= GbxLightFromMap_ScaleAD_TransDA.xy;
	LightFromMap.rg+= GbxLightFromMap_ScaleAD_TransDA.wz;	
	LAmbient *= LightFromMap.r;
#ifdef DGbxPDefinedPC3_Shadow
	fpart3	DirIntens = ShadowDiffus;
#else
	fpart3	DirIntens = LightFromMap.g;
#endif
	LDiffus = v.Color1*DirIntens;
	SpecSparkles *= DirIntens;
	SpecVarnish *= DirIntens;
	EnvCubic *= LightFromMap.r;
#else
	LightFromMap*= GbxLightFromMap_ScaleRGB_TransRGB.x;
	LightFromMap+= GbxLightFromMap_ScaleRGB_TransRGB.y;
	LDiffus *= LightFromMap;
	LAmbient *= 0.5 + 0.5*saturate(LightFromMap);
#endif
#endif

//	DiffuseGloss.a 		= 1; 
//	DiffuseGloss.rgb 	= DiffuseGloss*0.0001 +1 ;

	fpart 	Gloss 	= saturate(2*DiffuseGloss.a);
	fpart 	Metal 	= saturate(2*DiffuseGloss.a-1);
	Fresnel  	  	= Fresnel*Gloss;
	fpart LFresnel	= Fresnel*(1-LDiffus.r)*EnvCubic.a;	

//Metal = 0.1;

	float3	LHdr = LAmbient+LDiffus*(1-Metal);
	float	IHdr = dot(float3(1.f/3,1.f/3,1.f/3),LHdr);
	IHdr = log(IHdr)/log(2);
//	LHdr*= log(IHdr)/log(1.5)/IHdr;

	Output.Color.a		= 0;
//	Output.Color.rgb	= DiffuseGloss*(1-LFresnel)*(LAmbient*(Metal)+LDiffus*(1-Metal));

	Output.Color.rgb	= DiffuseGloss*(1-LFresnel)*LHdr;

	Output.Color.rgb   += DiffuseGloss*Metal*SpecSparkles*ShadowDiffus;
	Output.Color.rgb   += Fresnel*(EnvCubic);
	fpart3	LSpecular = SpecVarnish*ShadowSpecular;
	Output.Color.rgb   += Gloss*LSpecular;
	Output.Color.rgb   *= 2*Clouds;

#ifdef DGbxEditor
#define MDebug(__Value) Output.Color.rgb = Output.Color.rgb*0.001 + (__Value);
	MDbg_Above1(			Output.Color.rgb-1);
	MDbg_Clouds(			Clouds);
	MDbg_Diffuse(			LDiffus*DiffuseGloss);
	MDbg_Specular(			Gloss*LSpecular);
	MDbg_Occlusion(			VDepAmbOcc);
	MDbg_MapDiffuse(		DiffuseGloss);
	MDbg_MapSpecular(		Gloss*SpecVarnish*ShadowSpecular);
	MDbg_MapShadow0(		ShadowSpecular);
#if DGbxPDefinedPC3_Shadow
	MDbg_MapperShadow0(		DGbxPGenCodePC3_ShadowDbgMapper0);
#endif
	MDbg_LightDiffuse(		LHdr);
	MDbg_LightDiffuseAmb(	LAmbient);
	MDbg_LightDiffuseDir0(	LDiffus*(1-Metal));
	MDbg_LightSpecular(		LSpecular);
#endif
///////// DEBUG //////////////////
//	Output.Color.rgb	= Output.Color.rgb*0.001 + IHdr ;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LHdr ;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LAmbient;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LDiffus;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LDiffus*(1-Metal);
//	Output.Color.rgb	= Output.Color.rgb*0.001 + (fpart3)GbxLightAmbient;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + (fpart3)GbxLightAmbient+LDiffus;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + (fpart3)GbxLightAmbient+LDiffus*(1-Metal);
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LDiffus*DiffuseGloss;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LDiffus*DiffuseGloss*(1-Fresnel);
//	Output.Color.rgb	= Output.Color.rgb*0.001 + DiffuseGloss;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + DiffuseGloss.a;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + SpecSparkles;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + SpecVarnish;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + DiffuseGloss*Metal*SpecSparkles;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + texCUBE(MapEnvCubic, ReflecInW);
//	Output.Color.rgb	= Output.Color.rgb*0.001 + texCUBE(MapEnvCubic, ReflecInW)*DiffuseGloss.a*GbxLightDirRgb0;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + texCUBE(MapEnvCubic, ReflecInW)*DiffuseGloss.a*GbxLightDirRgb0*Fresnel;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + Fresnel;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + Gloss;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + Metal;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + LightFromMap;
//	Output.Color.rgb	= Output.Color.rgb*0.001 + tex2D(MapClouds, v.TcClouds);
//	Output.Color.rg	= Output.Color.rgb*0.001 + (abs(v.TcClouds)<0.1) + (abs(v.TcClouds-1)<0.1);
	return Output;
}

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);}
ps_2_a PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);}
