#include <Common.Phlsl.txt>
struct PS_INPUT		{
	float2 TcLeft	: TEXCOORD0;
	float2 TcRight	: TEXCOORD1;
};

struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

sampler2D	MapLeft;
sampler2D	MapRight;

const fpart In01GrayToFullColor = 0.5f;

fpart3 GrayToFull(fpart3 _Color)
{
	fpart  Lumi = dot(_Color, fpart3(1/3.0, 1/3.0, 1/3.0));
	return lerp(Lumi, _Color, In01GrayToFullColor);
}

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	Output.Color	= fpart4(1,0,0,1);

	fpart4 Left = tex2D(MapLeft, v.TcLeft);
	fpart4 Right= tex2D(MapRight,v.TcRight);

	fpart3 Red = GrayToFull(Left);
	fpart3 Cyan = GrayToFull(Right);

	Output.Color.r	= Red.r;
	Output.Color.gb	= Cyan.gb;
//	Output.Color.a  = 0.5*(Left.a+Right.a);
	Output.Color.a  = 1;
	return Output;
}
