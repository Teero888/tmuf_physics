sampler2D	MapColor;
sampler2D	MapDepth_Motion_MotionLen;
sampler2D	MapMotionLenInvLen;
sampler2D	MapShadow;

#include <Common.PHlsl.txt>
struct PS_INPUT		{
	float2 TexCoord	: TEXCOORD0;
};
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

#define MapColorAlphaIsOne

//#define DGbxPMrtMotionBlurPC3_TapCount 13	// With TapWeight=1	(28_tex 58_arith)
#define DGbxPMrtMotionBlurPC3_TapCount 6	// With TapWeight=TapLen*CenterILen + InputColor.a=? (26_tex 54_arith)
//#define DGbxPMrtMotionBlurPC3_TapCount 6	// With TapWeight=TapLen*CenterILen + InputColor.a=1 (26_tex 43_arith)

fpart	MotionMaxLen = 1;

static fpart2	CenterMotion;
static fpart	CenterLen;
static fpart	CenterILen;
static fpart4 	ColorSum;

static fpart2	CenterILL;

void AddContribution(float2 _TexCoord)
{
	// Fetch tap sample
	fpart4 	TapColor = tex2D(MapColor, _TexCoord);
	fpart4 	TapMotionLenILen = tex2D(MapMotionLenInvLen, _TexCoord);

	// Compute tap contribution
//	float 	TapWeight = 1;
//	float 	TapWeight = min(TapLen*CenterILen, TapILen*CenterLen); optim =
	float2	Prod = TapMotionLenILen*CenterILL;
	float 	TapWeight = min(Prod.x, Prod.y);
	// Accumulate color and contribution
#ifdef MapColorAlphaIsOne
	ColorSum 		+= TapColor * TapWeight;
#else
	ColorSum.rgb 	+= TapColor * TapWeight;
	ColorSum.a		+= TapWeight;
#endif
}

PS_OUTPUT Main(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;

	// Get center sample
	ColorSum = tex2D(MapColor, v.TexCoord);
	fpart4	Center_Depth_Motion_LenILen = tex2D(MapDepth_Motion_MotionLen, v.TexCoord);

	CenterMotion = Center_Depth_Motion_LenILen.yz;
	CenterLen  = Center_Depth_Motion_LenILen.w;
	CenterILen = 1.f/(CenterLen+0.001);

	CenterILL = fpart2(CenterILen, CenterLen);

#ifndef MapColorAlphaIsOne
	ColorSum.a = 1.0f;
#endif

	// Run through all taps
	for (int i = 0; i < DGbxPMrtMotionBlurPC3_TapCount; i++)
	{
		// Compute tap coordinates
		float2	TapOffset = CenterMotion * (1+i)/(1+DGbxPMrtMotionBlurPC3_TapCount);
		AddContribution(v.TexCoord + TapOffset);
		AddContribution(v.TexCoord - TapOffset);
	}

	// Normalize to get proper luminance
	Output.Color	= ColorSum / ColorSum.a;
 
#ifdef DFxDelayShadows
	fpart4	Shadow = tex2D(MapShadow, v.TexCoord);
	Output.Color	*= Shadow;
#endif

//////////////////////
//  Debug
//	Output.Color	= Output.Color*0.001 + Color;
//	Output.Color	= Output.Color*0.001 + CenterDepthBlur.x;
//	Output.Color	= Output.Color*0.001 + CenterDepthBlur.y;
//	Output.Color	= Output.Color*0.001 + abs(float4(CenterMotion.xy,0,1));
//	Output.Color	= Output.Color*0.001 + abs(float4(CenterMotion.xy,0,1))*500;
//	Output.Color	= Output.Color*0.001 + DepthBlur.y;
	return Output;
}

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);}
ps_2_a PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);}
