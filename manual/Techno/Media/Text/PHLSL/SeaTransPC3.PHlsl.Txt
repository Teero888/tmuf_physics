#include <Common.PHlsl.txt>

struct PS_INPUT {
    fpart4 Color0    	: COLOR0;
	float2 TcBump0  	: TEXCOORD0;
	float2 TcBump1  	: TEXCOORD1;
	float3 PosToEyeInW	: TEXCOORD2;
//	float2 TcHeightScale: TEXCOORD3;
	float4 TcShadowBuffer: TEXCOORD3;
};

struct PS_OUTPUT {
    fpart4 Color0 		: COLOR0;
};


// Samplers
sampler2D 	MapNormal;
sampler2D 	MapReflect;
sampler2D	MapFresnel;
sampler2D	MapRefrac;
sampler2D 	MapHeightScale;
samplerCUBE	MapCubeSxSySz;

// Tweaks
const fpart 	BumpScalePC3 = 1.f;
const fpart		RefracPertubPC3 = 0.1f;

// Gbx
const fpart4x3	GbxWorldPz01PlaneReflec;
const fpart4x3	GbxWorldPz01PlaneRefrac;

#define DMapFresnel

DGbxGenLoadPC3_PS_Shadow

PS_OUTPUT Main(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;

	fpart3		NormalInWorld = tex2D(MapNormal, v.TcBump0)+tex2D(MapNormal, v.TcBump1);
	NormalInWorld.xz *= BumpScalePC3;

// NVidia prefer tex lookup instead of nrm for performance, still
// their texture formats can't store normals in accurate way in 32b per texel
// only 4x16b=64b would be enough but cost a lot of memory (cubemap)
// => it only works for NormalInWorld, normalizing PosToEyeInW with a cube map
// => shows blocky artifacts
#ifdef DNormalize_UseCubeMap
	// NVidia GeForceFXs fallback
//	fpart3		PosToEyeInW = normalize(v.PosToEyeInW); Bug sur GeForceFXs !!
	float3		PosToEyeInW = normalize(v.PosToEyeInW);
	NormalInWorld = texCUBE(MapCubeSxSySz, NormalInWorld);
#else
	fpart3		PosToEyeInW = normalize(v.PosToEyeInW); // Should work on 6800
	NormalInWorld = normalize(NormalInWorld);
#endif

	// ReflecInCube normalization is not required (cube map)
	fpart3		ReflecInWorld = reflect(PosToEyeInW, NormalInWorld);
//	fpart3		ReflecInWorld = Reflec_NotNormalized(PosToEyeInW, NormalInWorld);

	fpart4		TcReflect = mul(ReflecInWorld, GbxWorldPz01PlaneReflec).xyzz;

	float3		RefracInWorld = -PosToEyeInW;
	RefracInWorld.xz += NormalInWorld.xz*RefracPertubPC3;
	// no need to normalize because the texcoord is projected !!
//	RefracInWorld = normalize(RefracInWorld);
	fpart4		TcRefrac = mul(RefracInWorld, GbxWorldPz01PlaneRefrac).xyzz;
	fpart4		Refrac = tex2Dproj(MapRefrac, TcRefrac);

/*	
	dot(normalize(PosToEyeInW), normalize(NormalInWorld)) <=>
	dot(PosToEyeInW, NormalInWorld)*
		rsqrt(dot(PosToEyeInW,PosToEyeInW)*dot(NormalInWorld,NormalInWorld))
	but second gains 3 instructions !
*/
	fpart		EyeDotN = dot(PosToEyeInW, NormalInWorld);
#ifdef DMapFresnel
	fpart		Fresnel = tex2D(MapFresnel, EyeDotN).r;
#else
	EyeDotN = saturate(EyeDotN);
	fpart		Fresnel = (1-EyeDotN)*(1-EyeDotN);
#endif

#if 0
	fpart4		Reflect = tex2Dproj(MapReflect, TcReflect);	
#else
	// Test : blur reflect with distant : cost 4 extra instr...
//	fpart		MipLodBias = 0.001*length(v.PosToEyeInW);
	fpart		MipLodBias = 0.000003*dot(v.PosToEyeInW,v.PosToEyeInW);
	// !! Warning : LodBias is uspported only in [-3,+3] range !!
	MipLodBias = min(MipLodBias, 2.7);
	fpart4		Reflect = tex2Dbias(MapReflect, float4(TcReflect.xyz/TcReflect.w,MipLodBias));
#endif

	DGbxGenCodePC3_PS_Shadow

	Output.Color0	= lerp(Refrac*ShadowDiffus, Reflect, Fresnel);
	Output.Color0  *= v.Color0*VDepAmbOcc;
	Output.Color0.a = 0;
#ifdef DGbxEditor
#define MDebug(__Value) Output.Color0.rgb = Output.Color0.rgb*0.001 + (__Value);
	MDbg_Above1(			Output.Color0.rgb-1);
	MDbg_NormalInW(			0.5+0.5*NormalInWorld);
	MDbg_Specular(			0);
	MDbg_Occlusion(			VDepAmbOcc);
	MDbg_Reflect(			Reflect*Fresnel);
	MDbg_Refract(			Refrac*ShadowDiffus*(1-Fresnel));
	MDbg_Fresnel(			Fresnel);
	MDbg_MapSpecular(		1);
	MDbg_MapShadow0(		ShadowSpecular);
#if DGbxPDefinedPC3_Shadow
	MDbg_MapperShadow0(		DGbxPGenCodePC3_ShadowDbgMapper0);
#endif
#endif
///////////////////////////
//  Debug
//	Output.Color0  += float4(EyeInCube,1);
//	Output.Color0.rgb	= Output.Color0*0.001 + HeightScale;
//	Output.Color0.rgb	= Output.Color0*0.001 + tex2D(MapHeight, v.TcHeight);
//	Output.Color0.rgb	= Output.Color0*0.001 + NormalInTgt;
//	Output.Color0.rgb	= Output.Color0*0.001 + NormalInWorld;
//	Output.Color0.rgb	= Output.Color0*0.001 + TcReflect;
//	Output.Color0.rgb	= Output.Color0*0.001 + EyeDotN;    
//	Output.Color0.rgb	= Output.Color0*0.001 + 0.5+0.5*EyeDotN;    
//	Output.Color0.rgb	= Output.Color0*0.001 + ReflecInWorld;
//	Output.Color0.rgb	= Output.Color0*0.001 + GbxWorldPz01PlaneReflec[1];
//	Output.Color0.rgb	= Output.Color0*0.001 + Fresnel;    
//	Output.Color0.rgb	= Output.Color0*0.001 + Reflect;
//	Output.Color0.rgb	= Output.Color0*0.001 + Refrac;
//	Output.Color0.rgb	= Output.Color0*0.001 + ShadowDiffus;
	return Output;    
}

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);}
ps_2_a PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);}
