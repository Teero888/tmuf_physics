#include <Common.Phlsl.txt>
struct PS_INPUT		{
	fpart4 Color0		: COLOR0;
	fpart4 Color1		: COLOR1;
	float3 TcSpecular 	: TEXCOORD0;
	float2 TcClouds		: TEXCOORD1;
	float4 TcShadow		: TEXCOORD2;
};
#define TcShadowBuffer TcShadow
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

samplerCUBE	MapSpecular;
sampler2D 	MapClouds;

DGbxPGenLoadPC3_Shadow

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	fpart3		Specular = texCUBE(MapSpecular, v.TcSpecular);
	fpart3		Clouds = tex2D(MapClouds, v.TcClouds);
	DGbxPGenCodePC3_Shadow
	fpart4		Light    = lerp(v.Color0,v.Color1,ShadowDiffus);
	Output.Color.rgb	= Light;
	Output.Color.rgb   += Specular*ShadowSpecular;
	Output.Color.rgb   *= Clouds*2;
	Output.Color.a		= 0;
	return Output;
}
