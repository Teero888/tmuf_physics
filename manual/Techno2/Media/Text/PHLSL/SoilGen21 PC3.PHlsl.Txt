#include <Common.Phlsl.txt>
struct PS_INPUT		{
	float4 Color0		: COLOR0;
	float4 Color1		: COLOR1;
	float2 TcBlend1		: TEXCOORD0;
	float2 TcBlend2		: TEXCOORD1;
	float2 TcBlendI		: TEXCOORD2;
	float2 TcClouds		: TEXCOORD3;
	float2 TcOcclusion	: TEXCOORD4;
	float2 TcPreLightGen	: TEXCOORD5;
	float4 TcShadowBuffer: TEXCOORD6;
};
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

sampler2D	MapBlend1;
sampler2D	MapBlend2;
sampler2D	MapBlendI;
sampler2D	MapClouds;
sampler2D	MapOcclusion;
sampler2D	MapPreLightGen;
samplerCUBE	MapSpecularCube;

const float4	GbxLightGenP_ScaleAD_TransDA;

DGbxPGenLoadPC3_Shadow

// DTwkPSkipMap_PreLightGen	(not parsed in includes)
// DTwkPSkipMap_Clouds 		(not parsed in includes)
// DTwkPSkipMap_Occlusion 	(not parsed in includes)
// DGbxPDefinedPC3_IsNight 	(not parsed in includes)
// DGbxPGenCodePC3_Shadow 	(not parsed in includes)

PS_OUTPUT Main(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;
	Output.Color	= fpart4(1,0,0,1);
	fpart4		Blend1 = tex2D(MapBlend1, v.TcBlend1);
	fpart4		Blend2 = tex2D(MapBlend2, v.TcBlend2);
	fpart		BlendI = tex2D(MapBlendI, v.TcBlendI).x;
	fpart4		Soil  = lerp(Blend1, Blend2, BlendI);
	fpart4		Clouds = tex2D(MapClouds, v.TcClouds);
	fpart		Occlusion = tex2D(MapOcclusion, v.TcOcclusion).x;

	DGbxPGenCodePC3_Shadow
#ifdef DTwkPSkipMap_PreLightGen
	fpart3		Light = lerp(v.Color0,v.Color1,ShadowDiffus);
#else
	fpart4		PreLightGen = tex2D(MapPreLightGen, v.TcPreLightGen);
	fpart3		Light = PreLightGen;
#ifndef DGbxPDefinedPC3_IsNight
	float3		CubeAmbientY = float3(0.52,0.6,0.7);
	Light.rg = Light.rg*GbxLightGenP_ScaleAD_TransDA.xy+GbxLightGenP_ScaleAD_TransDA.wz;
	fpart3		LDirIntens;
#ifdef DGbxPDefinedPC3_Shadow
	LDirIntens = ShadowDiffus;
#else
	LDirIntens = Light.g;
#endif	// combination don't work well, especially because of cars
	Light = saturate(Light.r*CubeAmbientY + LDirIntens*v.Color0);
#endif
#endif
	Output.Color.rgb = (Light*Soil).rgb;
#ifndef DTwkPSkipMap_Occlusion
	Output.Color.rgb*= Occlusion;
#endif
#ifndef DTwkPSkipMap_Clouds
	Output.Color.rgb*= Clouds.rgb*2;
#endif
	Output.Color.a	= 1;

/////////////////
// Debug
#ifdef DGbxEditor
#define MDebug(__Value) Output.Color.rgb = Output.Color.rgb*0.001 + (__Value);
#ifndef DTwkPSkipMap_Clouds
	MDbg_Clouds(			Clouds);
#endif
	MDbg_Diffuse(			Light*Soil);
	MDbg_Specular(			0);
	MDbg_Fresnel(			0);
	MDbg_NormalInW(			0.5+0.5*float3(0,1,0));
	MDbg_MapDiffuse(		Soil);
	MDbg_MapSpecular(		0);
	MDbg_MapOcclusion(		1);
	MDbg_LightDiffuse(		Light);
#ifdef DTwkPSkipMap_Occlusion
	Occlusion = 1;
#endif
#if !defined(DTwkPSkipMap_PreLightGen) && !defined(DGbxPDefinedPC3_IsNight)
	MDbg_LightDiffuseAmb(	PreLightGen.r*CubeAmbientY);
	MDbg_LightDiffuseDir0(	PreLightGen.g*v.Color0);
	MDbg_Occlusion(			PreLightGen.r*Occlusion);
#else
	MDbg_Occlusion(			Occlusion);
#endif
#endif

/////////////////////////
//  Debug
//	Output.Color	=  Output.Color*0.001 + Blend1;
//	Output.Color	=  Output.Color*0.001 + Blend2;
//	Output.Color	=  Output.Color*0.001 + BlendI;
//	Output.Color	=  Output.Color*0.001 + Blend;
//	Output.Color	=  Output.Color*0.001 + Clouds;
//	Output.Color	=  Output.Color*0.001 + Light;
//	Output.Color	=  Output.Color*0.001 + v.Color0;
//	Output.Color	=  Output.Color*0.001 + tex2D(MapPreLightGen, v.TcPreLightGen);
//	Output.Color	=  Output.Color*0.001 + tex2D(MapPreLightGen, v.TcPreLightGen).r;
	return Output;
}
ps_2_0 PS_OUTPUT psMain(const PS_INPUT v) {return Main(v);} 
glslf PS_OUTPUT psMain(PS_INPUT v) {return Main(v);}
