#include <Common.Phlsl.txt>
struct PS_INPUT		{
	float2 TcAdvert	: TEXCOORD0;
	float2 TcMask	: TEXCOORD1;
	float2 TcClouds	: TEXCOORD2;
	float4 TcShadow	: TEXCOORD3;
};
#define TcShadowBuffer TcShadow
struct PS_OUTPUT		{
	fpart4 Color	: COLOR0;
};

sampler2D MapAdvert;
sampler2D MapMask;
sampler2D MapClouds;

DGbxPGenLoadPC3_Shadow

ps_2_0 PS_OUTPUT psMain(const PS_INPUT v)
{
	PS_OUTPUT	Output = (PS_OUTPUT) 0;

	DGbxPGenCodePC3_Shadow
	fpart4	Advert	= tex2D(MapAdvert,	v.TcAdvert);
	fpart4	Mask 	= tex2D(MapMask,	v.TcMask);
	fpart4	Blend 	= lerp(Advert, Mask, Mask.a);
	fpart4	Clouds	= tex2D(MapClouds,	v.TcClouds);
	Output.Color	= Blend*Clouds*2*ShadowDiffus;
///////////////////
//  Debug
//	Output.Color.rgb= Output.Color*0.001 + Advert;
//	Output.Color.rgb= Output.Color*0.001 + Mask;
//	Output.Color.rgb= Output.Color*0.001 + Blend;
//	Output.Color.rgb= Output.Color*0.001 + Clouds;
//	Output.Color.rgb= Output.Color*0.001 + ShadowDiffus;
	return Output;
}
