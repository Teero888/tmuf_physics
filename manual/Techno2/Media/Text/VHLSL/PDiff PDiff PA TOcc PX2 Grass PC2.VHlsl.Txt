#include <Common.VHlsl.txt>

struct VS_INPUT		{
	float4 Position		: POSITION;
	float3 Normal		: NORMAL;
	float4 Color0		: COLOR0;
	float2 TcOcclusion	: TEXCOORD0;
};
struct VS_OUTPUT		{
	float4 HPosition	: POSITION;
	float  Fog			: FOG;
	float4 Color0		: COLOR0;
	float2 TcGrass		: TEXCOORD0;
	float2 TcClouds		: TEXCOORD1;
	float3 TcFresnel	: TEXCOORD2;
	float2 TcOcclusion	: TEXCOORD3;
};

const float4x4	GbxVisualPrCamera;
const float4x2	GbxVPositionToTexCoord0;
const float4x2	GbxVPositionToTexCoord1;
const float4	GbxEyeInVisual;

vs_1_1 VS_OUTPUT vsMain(const VS_INPUT v)
{
	VS_OUTPUT	Output = (VS_OUTPUT) 0;
#ifdef DGbxVDefinedPC3_Prelight
	DGbxVGenCodePC2_Prelight
#else
	Output.Color0.rgb  	 = LightAD_Dir1(v.Normal);
#endif
	Output.Color0.a  	 = 1;
	Output.HPosition	= mul(v.Position, GbxVisualPrCamera);
	Output.Fog 			= mul(v.Position, GbxVisualToFogTnL);
	Output.TcGrass		= mul(v.Position, GbxVPositionToTexCoord0);
	Output.TcClouds		= mul(v.Position, GbxVPositionToTexCoord1);
	Output.TcFresnel	=-(v.Position-GbxEyeInVisual).yzx;
	Output.TcOcclusion	= v.TcOcclusion;
	return Output;
}
